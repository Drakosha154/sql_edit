-- Таблица авторов
CREATE TABLE IF NOT EXISTS authors (
    author_id SERIAL PRIMARY KEY,
    full_name TEXT NOT NULL,
    birth_year INTEGER,
    country TEXT
);

-- Таблица жанров
CREATE TABLE IF NOT EXISTS genres (
    genre_id SERIAL PRIMARY KEY,
    name TEXT NOT NULL UNIQUE
);

-- Таблица книг
CREATE TABLE IF NOT EXISTS books (
    book_id SERIAL PRIMARY KEY,
    title TEXT NOT NULL,
    author_id INTEGER REFERENCES authors(author_id) ON DELETE CASCADE,
    genre_id INTEGER REFERENCES genres(genre_id) ON DELETE SET NULL,
    publication_year INTEGER,
    copies_available INTEGER DEFAULT 1 CHECK (copies_available >= 0),
    price DECIMAL CHECK (price >= 0)
);

-- Таблица читателей
CREATE TABLE IF NOT EXISTS readers (
    reader_id SERIAL PRIMARY KEY,
    full_name TEXT NOT NULL,
    email TEXT UNIQUE,
    registration_date DATE DEFAULT CURRENT_DATE,
    is_active BOOLEAN DEFAULT TRUE
);

-- Таблица выдачи книг (транзакции)
CREATE TABLE IF NOT EXISTS book_loans (
    loan_id SERIAL PRIMARY KEY,
    book_id INTEGER NOT NULL REFERENCES books(book_id) ON DELETE CASCADE,
    reader_id INTEGER NOT NULL REFERENCES readers(reader_id) ON DELETE CASCADE,
    loan_date DATE NOT NULL DEFAULT CURRENT_DATE,
    due_date DATE NOT NULL,
    return_date DATE,
);


Задания и решения
Задача 1: Базовый SELECT и JOIN
Описание: Выведите список всех книг с их названиями, именами авторов и годом публикации. Отсортируйте результат по году публикации (от новых к старым).

sql
SELECT b.title, a.full_name AS author_name, b.publication_year
FROM library.books b
JOIN library.authors a ON b.author_id = a.author_id
ORDER BY b.publication_year DESC;
Задача 2: Фильтрация и агрегация (WHERE и COUNT)
Описание: Найдите количество книг каждого автора, написанных в жанре 'Роман'. Выведите имя автора и количество его романов.

sql
SELECT a.full_name, COUNT(b.book_id) AS novels_count
FROM library.authors a
JOIN library.books b ON a.author_id = b.author_id
JOIN library.genres g ON b.genre_id = g.genre_id
WHERE g.name = 'Роман'
GROUP BY a.author_id, a.full_name;
Задача 3: Работа с датами и условием (WHERE с датами, IS NULL)
Описание: Выведите список читателей, которые взяли книги после 1 февраля 2024 года и еще не вернули их. Покажите имя читателя, название книги и дату выдачи.

sql
SELECT r.full_name, b.title, bl.loan_date
FROM library.book_loans bl
JOIN library.readers r ON bl.reader_id = r.reader_id
JOIN library.books b ON bl.book_id = b.book_id
WHERE bl.loan_date > '2024-02-01' AND bl.return_date IS NULL;
Задача 4: Группировка с условием на агрегат (GROUP BY и HAVING)
Описание: Найдите жанры, в которых средняя цена книги превышает 500 рублей. Выведите название жанра и среднюю цену, округленную до двух знаков.

sql
SELECT g.name, ROUND(AVG(b.price), 2) AS avg_price
FROM library.genres g
JOIN library.books b ON g.genre_id = b.genre_id
GROUP BY g.genre_id, g.name
HAVING AVG(b.price) > 500;
Задача 5: Подзапрос (IN)
Описание: Выведите имена авторов, у которых нет книг в наличии (copies_available = 0).

sql
SELECT full_name
FROM library.authors
WHERE author_id IN (
    SELECT DISTINCT author_id
    FROM library.books
    WHERE copies_available = 0
);
Задача 6: Оконная функция (RANK)
Описание: Для каждого автора выведите его самую дорогую книгу. Если у автора несколько книг с одинаковой максимальной ценой, выведите их все. В выводе: автор, название книги, цена, ранг.

sql
WITH ranked_books AS (
    SELECT
        a.full_name,
        b.title,
        b.price,
        RANK() OVER (PARTITION BY a.author_id ORDER BY b.price DESC) AS price_rank
    FROM library.authors a
    JOIN library.books b ON a.author_id = b.author_id
)
SELECT full_name, title, price
FROM ranked_books
WHERE price_rank = 1;
Задача 7: Объединение таблиц (UNION)
Описание: Создайте единый список всех людей, связанных с библиотекой: и авторов, и читателей. Укажите тип ('Автор' или 'Читатель') и отсортируйте по полному имени.

sql
SELECT full_name, 'Автор' AS person_type
FROM library.authors
UNION ALL
SELECT full_name, 'Читатель'
FROM library.readers
ORDER BY full_name;
Задача 8: Изменение данных (UPDATE)
Описание: Поднимите цену на 10% для всех книг жанра 'Фэнтези', опубликованных после 1990 года.

sql
UPDATE library.books
SET price = price * 1.10
WHERE genre_id = (SELECT genre_id FROM library.genres WHERE name = 'Фэнтези')
    AND publication_year > 1990;
-- Для проверки
SELECT * FROM library.books WHERE genre_id = 2;
Задача 9: Удаление данных (DELETE с использованием подзапроса)
Описание: Удалите всех неактивных читателей (is_active = FALSE), у которых нет текущих задолженностей по книгам (все книги возвращены).

sql
DELETE FROM library.readers
WHERE is_active = FALSE
    AND reader_id NOT IN (
        SELECT DISTINCT reader_id
        FROM library.book_loans
        WHERE return_date IS NULL
    );
-- Для проверки
SELECT * FROM library.readers;
Задача 10: Сложный запрос с оконной функцией и JOIN (топ-N)
Описание: Определите ТОП-3 авторов по общему количеству выданных их книг (учитывайте все записи в таблице book_loans). Выведите имя автора и количество выдач.

sql
SELECT
    a.full_name AS author_name,
    COUNT(bl.loan_id) AS total_loans
FROM library.authors a
JOIN library.books b ON a.author_id = b.author_id
JOIN library.book_loans bl ON b.book_id = bl.book_id
GROUP BY a.author_id, a.full_name
ORDER BY total_loans DESC
LIMIT 3;
