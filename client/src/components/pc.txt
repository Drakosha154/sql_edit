-- Создаем таблицу Product без внешних ключей
CREATE TABLE Product (
    maker TEXT NOT NULL,
    model TEXT PRIMARY KEY,
    type TEXT CHECK (type IN ('PC', 'Laptop', 'Printer'))
);

-- Создаем таблицу PC без внешнего ключа
CREATE TABLE PC (
    code INT PRIMARY KEY,
    model TEXT NOT NULL,
    speed INT NOT NULL,
    ram INT NOT NULL,
    hd INT NOT NULL,
    cd TEXT NOT NULL,
    price NUMERIC NOT NULL
);

-- Создаем таблицу Laptop без внешнего ключа
CREATE TABLE Laptop (
    code INT PRIMARY KEY,
    model TEXT NOT NULL,
    speed INT NOT NULL,
    ram INT NOT NULL,
    hd INT NOT NULL,
    price NUMERIC NOT NULL,
    screen INT NOT NULL
);

-- Создаем таблицу Printer без внешнего ключа
CREATE TABLE Printer (
    code INT PRIMARY KEY,
    model TEXT NOT NULL,
    color TEXT CHECK (color IN ('y', 'n')),
    type TEXT CHECK (type IN ('Laser', 'Jet', 'Matrix')),
    price NUMERIC NOT NULL
);

-- Добавляем внешние ключи через ALTER TABLE
ALTER TABLE PC ADD CONSTRAINT fk_pc_product
    FOREIGN KEY (model) REFERENCES Product(model);

ALTER TABLE Laptop ADD CONSTRAINT fk_laptop_product
    FOREIGN KEY (model) REFERENCES Product(model);

ALTER TABLE Printer ADD CONSTRAINT fk_printer_product
    FOREIGN KEY (model) REFERENCES Product(model);

-- Добавляем проверочные ограничения
ALTER TABLE Product ADD CONSTRAINT chk_product_type
    CHECK (type IN ('PC', 'Laptop', 'Printer'));

ALTER TABLE Printer ADD CONSTRAINT chk_printer_color
    CHECK (color IN ('y', 'n'));

ALTER TABLE Printer ADD CONSTRAINT chk_printer_type
    CHECK (type IN ('Laser', 'Jet', 'Matrix'));
